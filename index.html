function doPost(e){
 var d=JSON.parse(e.postData.contents);
 var ss=SpreadsheetApp.getActive();
 if(d.kind==='transfer'){
  var t=ss.getSheetByName('העברות')||ss.insertSheet('העברות');
  t.getRange('E:E').setNumberFormat('@');
  if(t.getLastRow()===0)t.appendRow(['תאריך','מס׳ העברה','מסניף','לסניף','ברקוד','מוצר','סוג ארגז','כמות בארגז','ארגזים','סה"כ','הערה']);
  d.rows.forEach(function(r){t.appendRow([d.date,d.no,d.from,d.to].concat(r).concat([d.note]))});
  return ContentService.createTextOutput('ok');
 }
 var s=ss.getSheetByName('ספירה')||ss.insertSheet('ספירה');
 s.getRange('D:D').setNumberFormat('@');
 if(s.getLastRow()===0)s.appendRow(['תאריך','סניף','סופר','ברקוד','מוצר','סוג','כמות בארגז','ארגזים','בודדים','סה"כ','במערכת','הפרש']);
 d.rows.forEach(function(r){s.appendRow([d.date,d.branch,d.who].concat(r))});
 return ContentService.createTextOutput('ok');
}
