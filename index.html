 s.getRange('D:D').setNumberFormat('@');
 if(s.getLastRow()===0)s.appendRow(['תאריך','סניף','סופר','ברקוד','מוצר','סוג','כמות בארגז','ארגזים','בודדים','סה"כ','במערכת','הפרש']);
